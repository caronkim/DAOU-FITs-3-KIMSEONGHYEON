## 실습문제 1

```java
package first;

import java.sql.*;
import java.util.Scanner;

public class BookSearchCommand {
    public static void main(String[] args) {
        Connection con = null;  // Connection 객체 초기화
        Scanner sc = new Scanner(System.in);    // 사용자 입력을 위한 Scanner 객체 생성
        System.out.print("검색어를 입력하세요 => ");
        String keyword = sc.nextLine();  // 사용자 입력
        try {
            // 1. Oracle Driver Loading
            Class.forName("oracle.jdbc.driver.OracleDriver");
            System.out.println("> Oracle Driver Loading 성공");

            // 2. Connection
            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##JAVA1";
            String pw = "1234";
            con = DriverManager.getConnection(url, id, pw); // DB 연결 생성
            System.out.println("> DB 연결 성공");

            // 3. SQL 작성
            String sql = "select bisbn, btitle, bauthor, bprice from book ";
            sql += "where btitle like ?";   // 동적 SQL 활용

            // 4. Statement 생성
            PreparedStatement pstmt = con.prepareStatement(sql);    // PreparedStatement -> 코드 작성 용이 / 빠른 속도 / 보안
            pstmt.setString(1, "%" + keyword + "%"); // 동적 쿼리에 값 삽입, setString()의 index는 1부터 시작
            System.out.println("> Statement 생성");

            // 5. 결과 처리
            // select문에서는 executeQuery()로 쿼리를 실행해야 함.
            ResultSet result = pstmt.executeQuery();    // executeQuery()가 성공적으로 완료하면 ResultSet 반환
            System.out.println("> DB 조회 완료\n");
            // 결과값이 있다면 반복문을 돌면서 출력, next()는 true, false 반환
            while (result.next()) {
                System.out.println(result.getString("bisbn") + " | "
                + result.getString("btitle") + " | "
                + result.getString("bauthor") + " | "
                +result.getString("bprice") + " | ");
            }

            // 6. 리소스 해제(역순으로 해제)
            result.close();     // 1. ResultSet 해제
            pstmt.close();      // 2. PreparedStatement 해제
            con.close();        // 3. Connection 해제
            System.out.println("\n> 리소스 해제 완료");

        } catch (ClassNotFoundException e) {
            // Class.forName()에서 해당하는 Driver가 없으면 ClassNotFoundException 객체 반환
            System.out.println("Oracle Driver를 찾을 수 없습니다.");
        } catch (SQLException e) {
            // Connection 형성 과정에서 에러 발생 시 SQLException 객체 반환
            System.out.println("DB 연결 과정에서 문제가 발생했습니다. " + e.getMessage());
        } catch (Exception e) {
            System.out.println("예기치 않은 문제가 발생했습니다." + e.getMessage());
        } finally {
            System.out.println("> 프로그램 종료");
        }

    }
}

```

## 실습 2번

```java
package first;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;

public class TransactionTest {
    public static void main(String[] args) {
        // 기본적인 JDBC 코드를 생성
        Connection con = null;
        try {
            // 1. Driver Loading
            Class.forName("oracle.jdbc.driver.OracleDriver");

            // 2. Connection 연결
            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##JAVA1";
            String pw = "1234";
            con = DriverManager.getConnection(url, id, pw); // memory와 시간 소모가 큰 작업.
            // 기본적으로 autoCommit mode로 Connection이 생성됨
            // 만약 transaction을 설정하려면
             con.setAutoCommit(false);   // transaction 시작
            // 명시적으로 com.commit() 혹은 con.rollback()이 호출되면 transaction 종료됨
            // 추가적으로 만약 해당 구문이 나오지 않고, connection이 정상 종료되면(리소스 해제 이후)
            // con.commit()으로 간주

            // 3. SQL 구문 작성
            String sql = "insert into memberTbl values(?,?)";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "2");
            pstmt.setString(2, "신사임당");

            // 4. SQL 구문 실행
            int result = pstmt.executeUpdate(); // return 값은 영향 받은 row의 수

            // 5. 결과 처리
            if (result == 1) {
                System.out.println("잘 처리되었어요.");
            }
            // transaction 끝
//            con.rollback(); // 지금까지 했던 모든 sql 작업을 롤백, transaction 무효화!
            con.commit();   // 지금까지 했던 모든 sql 작업을 데이터베이스에 적용, transaction 종료

            // 6. 리소스 해제
            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```
