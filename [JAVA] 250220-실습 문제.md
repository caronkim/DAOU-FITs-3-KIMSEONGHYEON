function myFunc() {

    $("tbody > tr").remove();

    $.ajax({
        async: true,    // 비동기 방식 혹은 동기 방식인지 선택, 기본값은 비동기
        url: "http://www.kobis.or.kr/kobisopenapi/webservice/rest/boxoffice/searchDailyBoxOfficeList.json",
        type: "GET",
        data: {
            key: "93b76955d7e5701adc6ffa28f74b0020",
            targetDt: $("#selectedDate").val().split("-").join("")
        },
        dataType: "json",    // 서버가 보내주는 결과데이터가 JSON이라는 것을 의미
        success: function (result) {
            alert("호출 성공")
            // console.log(result.boxOfficeResult.dailyBoxOfficeList);
            result.boxOfficeResult.dailyBoxOfficeList.forEach(function (item) {
                // console.log(item);
                kakaoImgSearch(item)
            })
        },
        error: function () {
            alert("이상해요")
        }
    })
}

function kakaoImgSearch(item) {
    let img;
    let img_url;
    let tr;
    let td;
    let del;

    $.ajax({
        async: false,
        url: "https://dapi.kakao.com/v2/search/image",
        type: "GET",
        dataType: "json",
        headers: {
            Authorization: "KakaoAK " + "1eb743a8bd98638bb938646ba1b844a3"
        },
        data: {
            query: item.movieNm
        },
        success: function (result) {

            tr = $("<tr></tr>");
            tr.attr("id", item.rnum);
            $("tbody").append(tr);

            // 순위
            td = $("<td></td>");
            td.attr("class", "rank");
            td.text(item.rank);
            $(`#${item.rnum}`).append(td);

            // 이미지
            td = $("<td></td>");
            img = $("<img />");
            img_url = result.documents[0].image_url;
            img.attr("src", img_url);
            img.attr("alt", item.movieNm);
            img.attr("class", "img");
            img.attr("width", "90");
            img.attr("height", "120");
            td.append(img);
            $(`#${item.rnum}`).append(td);

            // 제목
            td = $("<td></td>");
            td.attr("class", "movieNm");
            td.text(item.movieNm);
            $(`#${item.rnum}`).append(td);

            // 누적관객수
            td = $("<td></td>");
            td.attr("class", "audiAcc");
            td.text(item.audiAcc);
            $(`#${item.rnum}`).append(td);

            // 영화 개봉일
            td = $("<td></td>");
            td.attr("class", "openDt");
            td.text(item.openDt);
            $(`#${item.rnum}`).append(td);

            // 삭제
            td = $("<td></td>");
            del = $("<input>");
            del.attr("type", "button");
            del.val("삭제")
            del.attr("id", "del" + item.rnum);
            del.on("click", function (e) {
                $(`#${$(this).attr("id").slice(3)}`).remove();
            })
            td.append(del);
            $(`#${item.rnum}`).append(td);
        },
        error: function () {
            alert("이미지 검색에 실패했습니다.")
        }
    })
}
