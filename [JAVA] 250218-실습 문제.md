## Login Servlet
```java
package org.example.booksearchservlet.controller;

import org.example.booksearchservlet.service.MemberMyBatisMVCService;
import org.example.booksearchservlet.service.MemberMyBatisMVCServiceImpl;
import org.example.booksearchservlet.vo.MemberVO;

import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.http.HttpSession;
import java.io.IOException;
import java.io.PrintWriter;

@WebServlet("/login")
public class Login extends HttpServlet {
    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        RequestDispatcher requestDispatcher = req.getRequestDispatcher("login.html");
        requestDispatcher.forward(req,resp);
    }

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        req.setCharacterEncoding("utf-8");
        resp.setContentType("text/html; charset=utf-8");
        String userId = req.getParameter("id");
        String userPw = req.getParameter("pw");
        MemberMyBatisMVCService service = new MemberMyBatisMVCServiceImpl();
        MemberVO member = service.login(userId, userPw);
        System.out.println(member);
        PrintWriter out = resp.getWriter();

        if(member != null) {
            // 로그인 성공
            // WAS에게 session 저장 공간을 요청
            HttpSession session = req.getSession();
            // req.getSession(false);
            // 기존에 할당한 session 객체가 이미 존재하면 가져오기
            // 없으면 새로 만들지 않고 null을 리턴

            session.setAttribute("member", member);
            // setAttribute를 이용해 값을 저장
            // getAttribute를 이용해 값을 꺼내옴

            // invalidate() : 세션을 무효화 시킴

            // sendRedirect() : 정적 파일 전달 혹은 경로로 이동

            RequestDispatcher rd =
                    req.getRequestDispatcher("community.jsp");
            req.setAttribute("member", member);
            rd.forward(req, resp);

        } else {
            // 로그인 실패
            resp.sendRedirect("loginError.html");
        }

        out.flush();
        out.close();



    }

    @Override
    public void init() throws ServletException {
        super.init();
    }
}
```

## Community Service Implementation
```java
package org.example.booksearchservlet.service;

import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.example.booksearchservlet.dao.CommunityDAO;
import org.example.booksearchservlet.mybatis.MyBatisSessionFactory;
import org.example.booksearchservlet.vo.CommunityVO;

import java.util.List;

public class CommunityMyBatisMVCServiceImpl implements CommunityMyBatisMVCService{
    @Override
    public List<CommunityVO> selectAll() {
        SqlSessionFactory sqlSessionFactory = MyBatisSessionFactory.getSqlSessionFactory();
        SqlSession sqlSession = null;
        List<CommunityVO> list = null;

        // Session 생성
        try {
            // Auto Commit 해제
            sqlSession = sqlSessionFactory.openSession(false);
            System.out.println("세션 생성");
        } catch (Exception e) {
            e.printStackTrace();
        }

        // Transaction
        try {
            System.out.println("트랜잭션 시작");
            CommunityDAO dao = new CommunityDAO(sqlSession);
            list = dao.selectAll();
            sqlSession.commit();
        } catch (Exception e) {
            e.printStackTrace();
            sqlSession.rollback();
        } finally {
            sqlSession.close();
            System.out.println("트랜잭션 성공");
        }

        return list;
    }
}
```

```java
package org.example.booksearchservlet.service;

import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.example.booksearchservlet.dao.BookDAO;

        SqlSessionFactory sqlSessionFactory = MyBatisSessionFactory.getSqlSessionFactory();
        SqlSession sqlSession = null;
        MemberVO member = null;

        // Session 생성
        try {
            // Auto Commit 해제
            sqlSession = sqlSessionFactory.openSession(false);
            System.out.println("세션 생성");
        } catch (Exception e) {
            e.printStackTrace();
```

```xml
<!DOCTYPE configuration
        PUBLIC "-//mybatis.org/DTD Config 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-config.dtd" >

    <!-- JDBC Driver 클래스와 데이터베이스 연결 -->

    <!-- 별도의 외부파일에 작성해 해당 파일을 property로 가져와 사용 -->
    <properties resource="driver.properties"/>

    <!-- MyBatis 실행 설정 -->
    <!-- 상당히 많은 설정이 있지만, 대부분 default setting으로 이용 -->
    <settings>
        <!-- 데이터가 없는 경우 null로 표현하겠다 -->
        <setting name="jdbcTypeForNull" value="NULL"/>
        <setting name="logImpl" value="STDOUT_LOGGING"/>
    </settings>

    <!-- Type Alias가 나올 수 있음 -->
    <!-- 일반적으로 VO에 대한 alias가 설정됨 -->
    <typeAliases>
        <typeAlias alias="Book" type="org.example.booksearchservlet.vo.BookVO" />
        <typeAlias alias="BookDetail" type="org.example.booksearchservlet.vo.BookDetailVO" />
        <typeAlias alias="Member" type="org.example.booksearchservlet.vo.MemberVO"/>
        <typeAlias alias="Community" type="org.example.booksearchservlet.vo.CommunityVO"/>
    </typeAliases>

    <!-- DB 연결 정보 -->
    <environments default="development">
        <environment id="development">
            <!-- TransactionManager를 JDBC로 설정 = 수동으로 Transaction을 설정한다는 의미 -->
            <!-- commit, rollback을 코드에서 제어 => JDBC -->
            <!-- 만약 type이 MANAGED로 설정하면 자동으로 transation을 제어하게 할 수 있음 -->
            <!-- 단, Transaction Manager가 있어야 함 => Spring -->
            <transactionManager type="JDBC" />

            <!-- connection pool을 사용할지 여부를 설정 가능 -->
            <dataSource type="POOLED">
                <!-- 외부 변수 삽입 시 ${} 로 import -->
                <property name="driver" value="${db.driver}"/>
                <property name="url" value="${db.url}"/>
                <property name="username" value="${db.username}"/>
                <property name="password" value="${db.password}"/>
<!--                <property name="defaultEncoding" value="UTF-8"/>-->
            </dataSource>
        </environment>
    </environments>

    <!-- SQL 구문을 작성하는 mapper라는 xml파일의 경로를 지정 -->
    <mappers>
        <mapper resource="sqlmap/Book.xml"/>
        <mapper resource="sqlmap/Member.xml"/>
```
