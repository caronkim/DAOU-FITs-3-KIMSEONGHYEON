## JDBC 작성 흐름

```java
package first;

import java.sql.*;

public class FirstConnect {
    public static void main(String[] args) {
        // JDBC는 크게 6개의 단계로 사용함
        // 1. (JDBC)Driver Loading
        //      - 프로그램에서 딱 1번만 수행하면 됨
        //      - Class.forName -> 이름을 가지고 클래스를 찾는다!
        //      - 없을 경우 ClassNotFoundException 발생 -> try-catch 사용해야 함.
        //      - ojdbc.oracle.jdbc.driver에 driver 존재

        // 2. 데이터베이스 연결
        //      - 만약 연결 성공하면 연결 성공의 증표로 객체가 생성
        //      - 객체의 Class는 java.sql.Connection
        //      - 연결 시 3가지 정보가 필요
        //          - 어떤 데이터베이스에 접속하는지에 대한 정보(문자열) => JDBC URL
        //              - JDBC URL은 정해진 형식이 있음
        //              - jdbc:<db이름> (공통) + :<db 의존성에 따라 정의>
        //              - jdbc:oracle:<연결방식 - thin(라이브러리, HW 의존적이지만 빠름), oci(driver, 느리지만 범용적)>
        //              - jdbc:oracle:thin:<접속컴퓨터:포트번호>:<접속할 데이터베이스>
        //          - ID/PW가 필요

        // 3. SQL을 작성 => Statement(SQL 문장) 생성

        // 4. Query 실행
        //      - executeUpdate() : SQL 구문이 insert, update, delete 계열일 때 사용
        //          - 리턴값은 SQL에 의해서 영향을 받은 행의 수
        //      - executeQuery() : SQL 구문이 select 인 경우 사용
        //          - 리턴값은 ResultSet
        //          - next()로 행 위치를 이동할 수 있음 -> true, false 반환
        //          - getString("column 명")으로 현재 포인터(행)의 데이터를 가져옴

        // 5. 결과 처리
        //      - executeUpdate() : SQL에 의해서 영향을 받은 행의 수 반환
        //      - executeQuery() : ResultSet 반환

        // 6. 리소스 해제
        //      - 지금 사용된 자원은
        //      - Connection, PreparedStatement, ResultSet
        //      - 자원 해제 안 할 시, 데이터베이스에서 문제가 생김(사용하지 않지만 메모리를 점유, 과부하 초래)
        //      - 명시적으로 사용하지 않는다는 것을 알려줘야 함
        //      - 사용된 자원을 해제(close()를 호출)해야 함
        //      - 자원 해제는 역순으로 해야 함

        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");   // 1. Driver Loading
            System.out.println("드라이버 로딩 성공!!");
            Connection con = null;  // 2. 데이터베이스 연결
                                    // Connection은 java.sql.Connection임

            // JDBC URL은 정해진 형식이 있음, 3가지 정보 필요
            String JDBC_URL = "jdbc:oracle:thin:@localhost:1521:xe";
            String ID = "C##JAVA1";
            String PW = "1234";

            // Driver 연결
            // DriverManager도 java.sql에 존재
            // getConnection -> static, 인스턴스 없이 바로 호출 가능
            // DB 접속 불가 혹은 계정 정보 불일치 등 다양한 예외상황 발생 가능
            // 에러 발생 시 SQLException 객체 반환
            con = DriverManager.getConnection(JDBC_URL, ID, PW);
            System.out.println("데이터베이스 연결 성공");

            // Statement 생성, java.sql.Statement 활용
            // 문장이 빈 상태로 먼저 Statement 생성, SQL 문장을 싣고 갈 Statement 객체
            // Statement는 크게 3개가 있음
            // 1. 일반 Statement => 단순, 기능상으로는 약간 부족. 잘 안씀
            // 2. PreparedStatement => 이걸 많이 씀.
            //                          - 코드 작성 용이
            //                          - 효율이 좋다(속도 빠름)
            //                          - 보안성 (SQL Injection 방지)
            // 3. CallableStatement => 데이터베이스 내에 함수를 만들 수 있는데, JAVA에서 해당 함수를 호출하기 위해 사용
            //                          - Stored Procedure를 호출하여 결과값을 받아오기 위한 용도
            String sql = "select * from BOOK";
            PreparedStatement pstmt = con.prepareStatement(sql);    // Connection을 통해 prepareStatement 호출(sql이 준비되어 있다는 의미)
                                                                    // 따라서 sql을 먼저 넣어야 함(일반 statement는 후에 추가함)
            // Query 실행
            ResultSet rs = pstmt.executeQuery();

            // 결과 처리
            // 처음에는 데이터를 가리키고 있지 않으므로 next() 최초 실행
            if(rs.next()) {
                // 결과가 존재하면, btitle의 데이터 가져옴
                System.out.println(rs.getString("btitle"));
            } else {
                System.out.println("검색 결과가 없습니다.");
            }

            // 리소스 해제
            rs.close();
            pstmt.close();
            con.close();

        } catch (ClassNotFoundException e) {
            System.out.println("Driver를 찾을 수 없어요!");
        } catch (SQLException e) {
            System.out.println("DB 연결에 문제가 있어요!");
        }

    }
}
```

```java
package first;

import java.sql.*;
import java.util.Scanner;

// 책 제목에 대한 키워드를 입력받아서
// 챡 제목에 해당 키워드를 포함하고 있는 책을 검색
// 해당 책에 대한 책 제목, 저자, 가격 정보를 출력
public class SecondConnect {
    public static void main(String[] args) {
        Connection con = null;
        try {
            // 1. JDBC Driver Loading
            Class.forName("oracle.jdbc.driver.OracleDriver");

            // 2. 데이터베이스 연결(Connection 생성)
 ;
            // JDBC URL은 정해진 형식이 있음, 3가지 정보 필요
            String jdbcURL = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##JAVA1";
            String pw = "1234";
            con = DriverManager.getConnection(jdbcURL, id, pw);

            Scanner s = new Scanner(System.in);
            System.out.print("검색어를 입력하세요 => ");
            String keyword = s.nextLine();

            // SQL 문장을 만들어야 해
            String sql = "SELECT btitle, bauthor, bprice ";
            sql += "FROM BOOK WHERE BTITLE LIKE ?";  // 동적 SQL문 => ? 사용
            PreparedStatement pstmt = con.prepareStatement(sql);
            // 아직 SQL 문장이 완성되지 않음 -> 이 상태로 실행 불가
            pstmt.setString(1, "%" + keyword + "%"); // -> ?에 들어가는 부분
            // 문장 완성, 실행
            ResultSet rs = pstmt.executeQuery();
            while(rs.next()) {
                System.out.println(rs.getString("btitle") + " | "
                        + rs.getString("bauthor") + " | "
                        + rs.getString("bprice"));
            }
            rs.close();
            pstmt.close();
            con.close();
        } catch (Exception e) {
            System.out.println("오류 발생" + e.getMessage());
        }

    }
}
```

- Transaction
    - JDBC를 이용해 설정하려면, Connection에 설정해야 함!
    - 작업의 최소 단위
        - 예시) 이체
            - select 잔액
            - select 계좌 존재? select 잔액
            - update 잔액 - 1000
            - update 잔액  + 1000
    - 단일 SQL 구문을 여러 개 논리적인 단위로 묶어 놓은 것
    - 독립적인 일, 기능 단위로 묶어 놓은 것
    - ACID를 보장하기 위해 존재
        - https://medium.com/@goinhacker/consistency-db6bb98e07eb
        - Atomicity (원자성) → All or Nothing
            - 중간에 작업이 중단되면 작업 직전 상태로 돌려줌
        - Consistency(일치성)
            - 미리 정의된 데이터 규칙으로만 넣을 수 있음
        - Isolation(독립성)
            - 한 작업이 점유 시 다른 작업이 건들 수 없도록 고립
        - Durability(영구성)
            - 모든 데이터가 데이터베이스에 저장되도록 보장

```java
package first;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;

public class TransactionTest {
    public static void main(String[] args) {
        // 기본적인 JDBC 코드를 생성
        Connection con = null;
        try {
            // 1. Driver Loading
            Class.forName("oracle.jdbc.driver.OracleDriver");
            // 2. Connection 연결
            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##JAVA1";
            String pw = "1234";
            con = DriverManager.getConnection(url, id, pw); // memory와 시간 소모가 큰 작업.
            // 기본적으로 autoCommit mode로 Connection이 생성됨
            // 만약 transaction을 설정하려면
             con.setAutoCommit(false);   // transaction 시작
            // 명시적으로 com.commit() 혹은 con.rollback()이 호출되면 transaction 종료됨
            // 추가적으로 만약 해당 구문이 나오지 않고, connection이 정상 종료되면(리소스 해제 이후)
            // con.commit()으로 간주
            // 3. SQL 구문 작성
            String sql = "insert into memberTbl values(?,?)";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "2");
            pstmt.setString(2, "신사임당");
            // 4. SQL 구문 실행
            int result = pstmt.executeUpdate(); // return 값은 영향 받은 row의 수
            // 5. 결과 처리
            if (result == 1) {
                System.out.println("잘 처리되었어요.");
            }
            // transaction 끝
//            con.rollback(); // 지금까지 했던 모든 sql 작업을 롤백, transaction 무효화!
            con.commit();   // 지금까지 했던 모든 sql 작업을 데이터베이스에 적용, transaction 종료
            // 6. 리소스 해제
            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```
