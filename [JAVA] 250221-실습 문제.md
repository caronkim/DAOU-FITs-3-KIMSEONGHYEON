    public void init() throws ServletException {
        super.init();
    }

        service.insert(comment);
        resp.sendRedirect("/community/post?postId=" + postId);
    }

    @Override
    protected void doPut(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        req.setCharacterEncoding("utf-8");
        HttpSession session = req.getSession();
        MemberVO member = (MemberVO) session.getAttribute("member");
        String writer = req.getParameter("memberId");
        if(member.getMemberId() == writer) {
            MovieAJAXCommentService service = new MovieAJAXCommentServiceImpl();
            String memberId = member.getMemberId();
            String postId = req.getParameter("postId");
            String content = req.getParameter("content");
            CommentVO comment = new CommentVO(postId, memberId, content);
            service.update(comment);
            resp.sendRedirect("/community?postId=" + postId);

    }

    @Override
    protected void doDelete(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        req.setCharacterEncoding("utf-8");

            MovieAJAXCommentService service = new MovieAJAXCommentServiceImpl();

    @Override
    public void init() throws ServletException {
        super.init();
    }

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        req.setCharacterEncoding("utf-8");

            } else {
                posts = service.selectAll();

            RequestDispatcher rd = req.getRequestDispatcher("community.jsp");
            req.setAttribute("posts", posts);
            rd.forward(req, resp);
        } else {
            resp.sendRedirect("login");

    private static SqlSessionFactory sqlSessionFactory;

    static {
        try {
            String resouce = "SqlMapConfig.xml";
            Reader reader = Resources.getResourceAsReader(resouce);

            if (sqlSessionFactory == null) {
                sqlSessionFactory =
                        new SqlSessionFactoryBuilder().build(reader);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public static SqlSessionFactory getSqlSessionFactory() { return sqlSessionFactory; }
}

package org.example.movieajaxprogram.service;

import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.example.movieajaxprogram.dao.PostLikeDAO;
import org.example.movieajaxprogram.mybatis.MyBatisSessionFactory;
import org.example.movieajaxprogram.vo.PostLikeVO;

public class MovieAJAXPostLikeServiceImpl implements MovieAJAXPostLikeService{
    @Override
    public void like(PostLikeVO postLike) {
        SqlSessionFactory sqlSessionFactory = MyBatisSessionFactory.getSqlSessionFactory();
        SqlSession sqlSession = null;
        try {
            sqlSession = sqlSessionFactory.openSession(false);
            PostLikeDAO postLikeDAO = new PostLikeDAO(sqlSession);
            if (postLikeDAO.select(postLike) != null) {
                postLikeDAO.delete(postLike.getPostId());
            } else {
                postLikeDAO.insert(postLike);
            }
            sqlSession.commit();
        } catch (Exception e) {
            e.printStackTrace();
            sqlSession.rollback();
        } finally {
            sqlSession.close();
        }
    }
}

        } catch (Exception e) {
            e.printStackTrace();
            sqlSession.rollback();
        } finally {
            sqlSession.close();
        }

    }

        } catch (Exception e) {
            e.printStackTrace();
            sqlSession.rollback();
        } finally {
            sqlSession.close();
