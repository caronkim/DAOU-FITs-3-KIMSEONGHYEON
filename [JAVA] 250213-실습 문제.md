```java
            book.setBisbn(event.getNewValue());
            service.update(book);
        });

        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        titleCol.setCellFactory(TextFieldTableCell.forTableColumn());
        titleCol.setOnEditCommit(event -> {
            BookMyBatisMVCService service =
                    new BookMyBatisMVCServiceImpl();
            BookVO book = event.getRowValue();
            book.setBtitle(event.getNewValue());
            service.update(book);
        });

        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));
        priceCol.setCellFactory(TextFieldTableCell.forTableColumn(new javafx.util.converter.IntegerStringConverter()));
        priceCol.setOnEditCommit(event -> {
            BookMyBatisMVCService service =
                    new BookMyBatisMVCServiceImpl();
            BookVO book = event.getRowValue();
            book.setBprice(event.getNewValue());
            service.update(book);
        });

        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));
        authorCol.setCellFactory(TextFieldTableCell.forTableColumn());
        authorCol.setOnEditCommit(event -> {
            BookMyBatisMVCService service =
                    new BookMyBatisMVCServiceImpl();
            BookVO book = event.getRowValue();
            book.setBauthor(event.getNewValue());
            service.update(book);
        });

        searchBtn.setOnAction(event -> {
            BookMyBatisMVCService service =
                    new BookMyBatisMVCServiceImpl();
            ObservableList<BookVO> books = service.searchByTitle(searchField.getText());
            tableView.setItems(books);

        deleteBtn.setOnAction(event -> {
            BookMyBatisMVCService service =
                    new BookMyBatisMVCServiceImpl();
            service.deleteByISBN(deleteIsbn);

                    new BookMyBatisMVCServiceImpl();

                    titleField.getText(),
                    Integer.parseInt(priceField.getText()),

            row.setOnMouseClicked(e -> {
                if (e.getClickCount() == 1 && row.getItem() != null) {
                    selectedBook = row.getItem();
                    deleteIsbn = selectedBook.getBisbn();
                }
            });
            return row;
        });
    }
```

```java
package assignment.dao;

import assignment.vo.BookVO;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import org.apache.ibatis.session.SqlSession;

import java.util.HashMap;
import java.util.List;

// 원래는 Connection이 제공되었음
// 이번엔 SqlSessionFactory를 DAO에게 전달!
public class BookDAO {

    private SqlSession sqlSession;

    // DI 처리
    public BookDAO(SqlSession sqlSession) {
        this.sqlSession = sqlSession;
    }

    // 모든 책 정보 조회
    public List<HashMap<String, Object>> selectAllHashMap() {
        List<HashMap<String, Object>> list = null;

        // 이 sqlSession에 transaction이 설정됨
        try {
            // selectOne -> select 하는데, row 하나만 갖고옴.
            // mapper에 명시된 select의 id를 명시하여 사용할 sql 쿼리 지정
            // [mapper의 namespace].[select의 id]
            // 두 번째 인자에 parameter 값을 전달
            list = sqlSession.selectList("assignment.MyBook.selectAllHashMap");
        } catch (Exception e) {
            e.printStackTrace();
        }
        return list;
    }

    public BookVO selectByISBNBookVO(String bisbn) {
        BookVO bookVO = null;
        // Session은 Factory에서 openSession() 메서드를 통해 가져옴

        // 이 sqlSession에 transaction이 설정됨
        try {
            // selectOne -> select 하는데, row 하나만 갖고옴.
            // mapper에 명시된 select의 id를 명시하여 사용할 sql 쿼리 지정
            // [mapper의 namespace].[select의 id]
            // 두 번째 인자에 parameter 값을 전달
            bookVO = sqlSession.selectOne("assignment.MyBook.selectByISBNBookVO", bisbn);
        } catch (Exception e) {
            e.printStackTrace();
        }
        return bookVO;
    }

    public ObservableList<BookVO> selectByTitleBookVO(String text) {
        List<BookVO> list = null;
        ObservableList<BookVO> books = FXCollections.observableArrayList();
        // 이 sqlSession에 transaction이 설정됨
        try {
            list = sqlSession.selectList("assignment.MyBook.selectByTitleBookVO", text);
            for(BookVO book : list) {
                books.add(book);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
        return books;
    }

    public void insert(BookVO bookVO) {
        try {
            sqlSession.insert("assignment.MyBook.insert",bookVO);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public void update(BookVO bookVO) {
        try {
            sqlSession.update("assignment.MyBook.update",bookVO);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public void delete(String bisbn) {
        try {
            sqlSession.delete("assignment.MyBook.delete",bisbn);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```

## SessionFactory
```java
package assignment.mybatis;

import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;

import java.io.Reader;

// MyBatis 사용, JDBC는 Connection을 통해 데이터베이스 처리
// MyBatis는 Connection이 아닌 Session을 사용!
// Session을 만드는 클래스를 하나 생성
// SQL Session이라는 객체가 DAO에서 필요
// 이 클래스를 통해 SQL Session 객체를 뽑아낼 수 있음
// SQL Session 객체를 이용해야만 우리가 SQL문을 실행할 수 있음(xml의 mapper파일)
public class MyBatisSessionFactory {
    private static SqlSessionFactory sqlSessionFactory;

    // 클래스 사용되면 바로 호출
    static {
        try {
            // Classpath가 지정되어 있기 때문에 이러한 경로로 설정 가능
            String resource = "./SqlMapConfig.xml";
            // Resource에 해당하는 설정 파일을 읽어들이는 과정
            Reader reader = Resources.getResourceAsReader(resource);

            // session이 null이면 만듦
            if(sqlSessionFactory == null) {
                // 이 때, connection pool과 같은 기본 세팅이 만들어짐!
                sqlSessionFactory =
                        new SqlSessionFactoryBuilder().build(reader);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // 만들어진 session 호출
    public static SqlSessionFactory getSqlSessionFactory() {
        return sqlSessionFactory;
    }
}
```

```java
package assignment.service;

import assignment.dao.BookDAO;
import assignment.mybatis.MyBatisSessionFactory;
import assignment.vo.BookVO;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;

public class BookMyBatisMVCServiceImpl implements BookMyBatisMVCService{
    @Override
    public ObservableList<BookVO> searchByTitle(String text) {
        SqlSessionFactory sqlSessionFactory = MyBatisSessionFactory.getSqlSessionFactory();
        SqlSession sqlSession = null;
        ObservableList<BookVO> list = FXCollections.observableArrayList();

        // Session 생성
        try {
            // Auto Commit 해제
            sqlSession = sqlSessionFactory.openSession(false);
            System.out.println("세션 생성");
        } catch (Exception e) {
            e.printStackTrace();
        }

        // Transaction
        try {
            System.out.println("트랜잭션 시작");
            BookDAO dao = new BookDAO(sqlSession);
            list = dao.selectByTitleBookVO("%" + text + "%");
            sqlSession.commit();
        } catch (Exception e) {
            e.printStackTrace();
            sqlSession.rollback();
        } finally {
            sqlSession.close();
            System.out.println("트랜잭션 성공");
        }

        return list;
    }
```
