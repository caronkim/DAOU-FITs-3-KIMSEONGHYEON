```java
    // 모든 책 정보 조회
    public List<HashMap<String, Object>> selectAllHashMap() {
        List<HashMap<String, Object>> list = null;

        // 이 sqlSession에 transaction이 설정됨
        try {
            // selectOne -> select 하는데, row 하나만 갖고옴.
            // mapper에 명시된 select의 id를 명시하여 사용할 sql 쿼리 지정
            // [mapper의 namespace].[select의 id]
            // 두 번째 인자에 parameter 값을 전달
            list = sqlSession.selectList("assignment.MyBook.selectAllHashMap");
        } catch (Exception e) {
            e.printStackTrace();
        }
        return list;
    }

    public List<BookVO> select(String keyword, String price) {
        Map<String,Object> map = new HashMap<String,Object>();
        List<BookVO> list = null;

        map.put("bprice", Integer.parseInt(price));

        try {
            list = sqlSession.selectList("bookSearch.MyBook.select", map);

        } catch (Exception e) {
            e.printStackTrace();
        }
        return list;
    }

    public BookDetailVO selectByISBN(String bisbn) {

        BookDetailVO book = null;

            book = sqlSession.selectOne("bookSearch.MyBook.selectByISBNBookVO", bisbn);

        } catch (Exception e) {
            e.printStackTrace();
        }
        return book;

        try {
            sqlSession.insert("bookSearch.MyBook.insert",bookVO);

            e.printStackTrace();
        }
    }

    public void update(BookVO bookVO) {

            e.printStackTrace();

    }

    public void delete(String bisbn) {

            e.printStackTrace();
        }
    }
```

## Session Factory
```java
package org.example.booksearchservlet.mybatis;

import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;

import java.io.Reader;

// MyBatis 사용, JDBC는 Connection을 통해 데이터베이스 처리
// MyBatis는 Connection이 아닌 Session을 사용!
// Session을 만드는 클래스를 하나 생성
// SQL Session이라는 객체가 DAO에서 필요
// 이 클래스를 통해 SQL Session 객체를 뽑아낼 수 있음
// SQL Session 객체를 이용해야만 우리가 SQL문을 실행할 수 있음(xml의 mapper파일)
public class MyBatisSessionFactory {
    private static SqlSessionFactory sqlSessionFactory;

    // 클래스 사용되면 바로 호출
    static {
        try {
            // Classpath가 지정되어 있기 때문에 이러한 경로로 설정 가능
            String resource = "SqlMapConfig.xml";
            // Resource에 해당하는 설정 파일을 읽어들이는 과정
            Reader reader = Resources.getResourceAsReader(resource);

            // session이 null이면 만듦
            if(sqlSessionFactory == null) {
                // 이 때, connection pool과 같은 기본 세팅이 만들어짐!
                sqlSessionFactory =
                        new SqlSessionFactoryBuilder().build(reader);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // 만들어진 session 호출
    public static SqlSessionFactory getSqlSessionFactory() {
        return sqlSessionFactory;
    }
}
```

```java
public class BookMyBatisMVCServiceImpl implements BookMyBatisMVCService{
    @Override
    public List<BookVO> search(String keyword, String price) {
        SqlSessionFactory sqlSessionFactory = MyBatisSessionFactory.getSqlSessionFactory();
        SqlSession sqlSession = null;
        List<BookVO> list = null;

        // Session 생성
        try {
            // Auto Commit 해제
            sqlSession = sqlSessionFactory.openSession(false);
            System.out.println("세션 생성");
        } catch (Exception e) {
            e.printStackTrace();
        }

        // Transaction
        try {
            System.out.println("트랜잭션 시작");
            BookDAO dao = new BookDAO(sqlSession);
            list = dao.select(keyword, price);
            sqlSession.commit();
        } catch (Exception e) {
            e.printStackTrace();
            sqlSession.rollback();
        } finally {
            sqlSession.close();
            System.out.println("트랜잭션 성공");
        }

        return list;
    }
```

```xml
        PUBLIC "-//mybatis.org/DTD Config 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-config.dtd" >
<configuration>
    <!-- JDBC Driver 클래스와 데이터베이스 연결 -->

    <!-- 상당히 많은 설정이 있지만, 대부분 default setting으로 이용 -->

        <setting name="jdbcTypeForNull" value="NULL"/>

    </settings>

    <!-- Type Alias가 나올 수 있음 -->
    <!-- 일반적으로 VO에 대한 alias가 설정됨 -->
    <typeAliases>
        <typeAlias alias="Book" type="org.example.booksearchservlet.vo.BookVO" />
        <typeAlias alias="BookDetail" type="org.example.booksearchservlet.vo.BookDetailVO" />
    </typeAliases>

    <!-- DB 연결 정보 -->
    <environments default="development">
        <environment id="development">
            <!-- TransactionManager를 JDBC로 설정 = 수동으로 Transaction을 설정한다는 의미 -->
            <!-- commit, rollback을 코드에서 제어 => JDBC -->
            <!-- 만약 type이 MANAGED로 설정하면 자동으로 transation을 제어하게 할 수 있음 -->
            <!-- 단, Transaction Manager가 있어야 함 => Spring -->
            <transactionManager type="JDBC" />

            <!-- connection pool을 사용할지 여부를 설정 가능 -->
            <dataSource type="POOLED">
```
