```java
package assignment;

import assignment.utils.DBFetch;
import assignment.vo.Book;
import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.input.MouseButton;

import java.net.URL;
import java.util.ResourceBundle;

public class BookTableFXController implements Initializable {

    @FXML
    private TableView<Book> tableView;

    @FXML
    private Button searchBtn;

    @FXML
    private TextField searchText;

    @FXML
    private TableColumn<Book, String> isbnCol;

    @FXML
    private TableColumn<Book, String> titleCol;

    @FXML
    private TableColumn<Book, Integer> priceCol;

    @FXML
    private TableColumn<Book, String> authorCol;

    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {

        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));

        searchBtn.setOnAction(event -> {
            try {
                ObservableList<Book> books = DBFetch.search(searchText.getText());
                tableView.setItems(books);
            } catch (Exception e) {
                System.out.println("DB 조회 오류" + e.getMessage());
            }
        });

        tableView.setRowFactory(tv_event -> {
            TableRow<Book> row = new TableRow<>();
            ContextMenu contextMenu = new ContextMenu();
            MenuItem deleteItem = new MenuItem("삭제");

            deleteItem.setOnAction(event -> {
                Book selectedBook = row.getItem();
                if (selectedBook != null) {
                    try {
                        DBFetch.delete(selectedBook.getBisbn()); // DB에서 삭제
                        tableView.getItems().remove(selectedBook); // UI에서 삭제
                    } catch (Exception e) {
                        System.out.println("DB 삭제 오류: " + e.getMessage());
                    }
                }
            });

            contextMenu.getItems().add(deleteItem);

            row.setOnMouseClicked(r_event -> {
                if (r_event.getButton() == MouseButton.SECONDARY && !row.isEmpty()) {
                    contextMenu.show(row, r_event.getScreenX(), r_event.getScreenY());
                } else {
                    contextMenu.hide();
                }

                } else {

                }
            });

            return row;
        });
    }
}
```

### DB 통신 클래스

```java
package assignment.utils;

import assignment.vo.Book;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

public class DBFetch {
    public static String db = "jdbc:oracle:thin:@localhost:1521:xe";
    public static String id = "C##JAVA1";
    public static String pw = "1234";

    public static ObservableList<Book> search(String keyword) throws Exception{
        Class.forName("oracle.jdbc.driver.OracleDriver");
        Connection con = null;

        con = DriverManager.getConnection(db, id, pw);

        String sql = "select bisbn, btitle, bprice, bauthor ";
        sql += "from book where btitle like ?";

        PreparedStatement pstmt = con.prepareStatement(sql);

        pstmt.setString(1, "%" + keyword + "%");

        ResultSet rs = pstmt.executeQuery();
        ObservableList<Book> books = FXCollections.observableArrayList();

        while (rs.next()) {
            books.add(new Book(rs.getString("bisbn"),
                    rs.getString("btitle"),
                    rs.getInt("bprice"),
                    rs.getString("bauthor")));
        }

        rs.close();
        pstmt.close();
        con.close();

        return books;
    }

    public static void update() {

    }

    public static void delete(String isbn) throws Exception{
        Class.forName("oracle.jdbc.driver.OracleDriver");
        Connection con = null;

        con = DriverManager.getConnection(db, id, pw);

        String sql = "delete from book ";
        sql += "where bisbn = ?";

        PreparedStatement pstmt = con.prepareStatement(sql);

        pstmt.setString(1, isbn);

        int result = pstmt.executeUpdate();

        if (result == 1) {
            System.out.println(isbn + "이 정상적으로 삭제되었습니다.");
        }

        pstmt.close();
        con.close();
    }
}
```
