```sql
DROP TABLE MEMBERTBL;

DROP TABLE POST;
DROP SEQUENCE POST_SEQ;

COMMIT;

CREATE TABLE MEMBERTBL(
    MEMBER_ID VARCHAR2(50) PRIMARY KEY,
    MEMBER_PASSWORD VARCHAR2(20),
    MEMBER_NAME VARCHAR2(10),
    MEMBER_PHONE VARCHAR2(15),
    MEMBER_EMAIL VARCHAR2(30),
    CREATED_AT DATE DEFAULT SYSDATE
);

SELECT * FROM MEMBERTBL;

CREATE TABLE POST(
    POST_ID VARCHAR2(50) PRIMARY KEY,
    TITLE VARCHAR2(100),
    CONTENT CLOB,
    MEMBER_ID VARCHAR2(50),
    VIEW_CNT NUMBER DEFAULT 0,
    CREATED_AT DATE DEFAULT SYSDATE
);

CREATE SEQUENCE POST_SEQ
START WITH 1
INCREMENT BY 1
NOCACHE;

COMMIT;

INSERT INTO POST(
POST_ID, TITLE, CONTENT, MEMBER_ID)
VALUES (POST_SEQ.nextval, '그린 한식 뷔페 짱', '제가 먹은 뷔페 중 제일 가성비 좋고 맛있어요!', 'user1');

INSERT INTO POST(
POST_ID, TITLE, CONTENT, MEMBER_ID)
VALUES (POST_SEQ.nextval, '르프리크 짱', '제가 먹은 치킨 버거 중 제일 가성비 좋고 맛있어요!', 'user2');

INSERT INTO POST(
POST_ID, TITLE, CONTENT, MEMBER_ID)
VALUES (POST_SEQ.nextval, '꾸아 짱', '제가 먹은 쌀국수 중 제일 가성비 좋고 맛있어요!', 'user3');

COMMIT;

UPDATE POST
SET VIEW_CNT = VIEW_CNT + 1
WHERE POST_ID = 1;

COMMIT;

SELECT * FROM POST;

CREATE TABLE POST_LIKE (
    post_like_id varchar2(50) PRIMARY KEY,
    member_id varchar2(50),
    post_id varchar2(50)
);

COMMIT;

CREATE SEQUENCE POST_LIKE_SEQ
START WITH 1
INCREMENT BY 1
NOCACHE;

COMMIT;

INSERT INTO POST_LIKE(
POST_LIKE_ID, POST_ID, MEMBER_ID)
VALUES (POST_LIKE_SEQ.nextval, '1', 'user3');

INSERT INTO POST_LIKE(
POST_LIKE_ID, POST_ID, MEMBER_ID)
VALUES (POST_LIKE_SEQ.nextval, '1', 'user2');

INSERT INTO POST_LIKE(
POST_LIKE_ID, POST_ID, MEMBER_ID)
VALUES (POST_LIKE_SEQ.nextval, '2', 'user3');

COMMIT;

SELECT * FROM POST_LIKE;

CREATE TABLE POST_COMMENT(
    POST_COMMENT_ID VARCHAR2(50) PRIMARY KEY,
    POST_ID VARCHAR2(50),
    MEMBER_ID VARCHAR2(50),
    COMMENT_CONTENT VARCHAR2(100),
    CREATED_AT DATE DEFAULT SYSDATE
);

CREATE SEQUENCE POST_COMMENT_SEQ
START WITH 1
INCREMENT BY 1
NOCACHE;

COMMIT;

INSERT INTO POST_COMMENT(
POST_COMMENT_ID, POST_ID, MEMBER_ID, COMMENT_CONTENT)
VALUES (POST_COMMENT_SEQ.nextval, '1', 'user1', '좋아요 많이 눌러주세요!');

INSERT INTO POST_COMMENT(
POST_COMMENT_ID, POST_ID, MEMBER_ID, COMMENT_CONTENT)
VALUES (POST_COMMENT_SEQ.nextval, '1', 'user2', '거기 지금 아껴두고 있습니다,,');

INSERT INTO POST_COMMENT(
POST_COMMENT_ID, POST_ID, MEMBER_ID, COMMENT_CONTENT)
VALUES (POST_COMMENT_SEQ.nextval, '1', 'user3', '좋아요 많이 눌러주세요!');

COMMIT;

SELECT * FROM POST_COMMENT;

SELECT P.*,
    (SELECT COUNT(*) FROM POST_COMMENT C WHERE P.POST_ID = C.POST_ID) COM_CNT,
    (SELECT COUNT(*) FROM POST_LIKE L WHERE P.POST_ID = L.POST_ID) AS LIKE_CNT
FROM POST P;
```
