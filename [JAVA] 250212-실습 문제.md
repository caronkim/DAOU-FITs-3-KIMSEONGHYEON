package diassignment.dao;

import diassignment.vo.BookVO;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

public class BookDAO {

    private Connection con;

    public BookDAO (Connection con) {
        this.con = con;
    }
    public ObservableList<BookVO> search(String text) {
        ObservableList<BookVO> books = FXCollections.observableArrayList();
        try {
            String sql = "select bisbn, btitle, bprice, bauthor ";
            sql += "from book where btitle = ?";
            PreparedStatement ps = con.prepareStatement(sql);
            ps.setString(1, "%" + text + "%");
            ResultSet rs = ps.executeQuery();
            while(rs.next()) {
                BookVO book = new BookVO(rs.getString("bisbn"),
                        rs.getString("btitle"),
                        rs.getInt("bprice"),
                        rs.getString("bauthor"));
                books.add(book);
            }
            rs.close();
            ps.close();
        } catch (Exception e) {
            e.printStackTrace();
        }

        return books;
    }

    public void delete(String text) {
        ObservableList<BookVO> books = FXCollections.observableArrayList();
        try {
            String sql = "delete from book where bisbn == ? ";
            PreparedStatement ps = con.prepareStatement(sql);
            ps.setString(1, "%" + text + "%");
            int cnt = ps.executeUpdate();
            if (cnt != 1) {
                throw new Exception("DB 삭제 오류가 발생했습니다.");
            }
            ps.close();
        } catch (Exception e) {
            e.printStackTrace();
        }

    }

    public void update(String bisbn, String columnLabel, String text) {
        ObservableList<BookVO> books = FXCollections.observableArrayList();
        try {
            String sql = "update book ";
            sql += "set ? = ? ";
            sql += "where bisbn = ?";
            PreparedStatement ps = con.prepareStatement(sql);
            ps.setString(1, columnLabel);
            ps.setString(2, text);
            ps.setString(3, bisbn);
            ps.executeUpdate();
            ps.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }

    }
}

package diassignment.service;

import diassignment.dao.BookDAO;
import diassignment.vo.BookVO;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;

import java.sql.Connection;
import java.sql.DriverManager;

public class BookMVCServiceImpl implements BookMVCService{
    @Override
    public ObservableList<BookVO> searchByKeyword(String text){
        // Service는 Business 로직을 처리(transaction)
        // 메서드 하나 = 트랜잭션
        // JDBC에서 Transaction은 Connection에서 설정
        ObservableList<BookVO> list = FXCollections.observableArrayList();
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");
            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##JAVA1";
            String pw = "1234";
            Connection con = DriverManager.getConnection(url, id, pw);
            con.setAutoCommit(false);   // Transaction 시작(select에서는 쓸 필요 없음. DB 변경 있을 때만)
            // 로직처리 1번
            // DB 처리 1번 => 단일 Database 처리(SQL문장 1개)
            // con.commit() or con.rollback()

            BookDAO bookDAO = new BookDAO(con); // injection
            list = bookDAO.search(text);

            // 실패 시 rollback, 성공 시 commit
            if (list != null) {
                con.commit();
            } else {
                con.rollback();
            }
            // Connection에 대한 close는 Service에서 이루어져야 함(Transaction이 끝나면 연결 종료)
            con.close();

        } catch (Exception e) {
            e.printStackTrace();
        }
        return list;
    }

    @Override
    public void deleteByIsbn(String text) {
        // Service는 Business 로직을 처리(transaction)
        // 메서드 하나 = 트랜잭션
        // JDBC에서 Transaction은 Connection에서 설정
        Connection con = null;
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");
            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##JAVA1";
            String pw = "1234";
            con = DriverManager.getConnection(url, id, pw);
            con.setAutoCommit(false);   // Transaction 시작(select에서는 쓸 필요 없음. DB 변경 있을 때만)
            // 로직처리 1번
            // DB 처리 1번 => 단일 Database 처리(SQL문장 1개)
            // con.commit() or con.rollback()

            BookDAO bookDAO = new BookDAO(con); // injection
            try {
                bookDAO.delete(text);
                con.commit();
            } catch (Exception e) {
                e.printStackTrace();
                con.rollback();
            }

            // Connection에 대한 close는 Service에서 이루어져야 함(Transaction이 끝나면 연결 종료)
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

                new FXMLLoader(getClass().getResource("bookmvc.fxml"));

            root = loader.load();
        } catch (Exception e) {
            e.printStackTrace();

        Scene scene = new Scene(root);

        stage.setTitle("도서 DB Application");

        launch(args);
    }
